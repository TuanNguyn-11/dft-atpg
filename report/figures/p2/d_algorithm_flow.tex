% Luồng khái niệm nhất quán với mã giả P2.
% Bố cục: cột chính x=0 (bước 1,3 cm), cột phụ x=3,6 và x=7,4; các đường
% vòng đi ngoài khối để không cắt chữ.
\begin{tikzpicture}[>=Stealth,
 every node/.style={font=\scriptsize,align=center},
 box/.style={draw,rounded corners,text width=2.3cm,minimum height=7mm},
 check/.style={draw,diamond,aspect=2.8,inner sep=1pt,text width=1.7cm},
 lbl/.style={font=\scriptsize,inner sep=2pt}]
% Cột chính
\node[box] (start) at (0,0) {Chọn PDCF đầu tiên};
\node[check] (limit) at (0,-1.3) {Hết giới hạn?};
\node[box] (imply) at (0,-2.6) {Implication tiến/lùi\\cập nhật frontier};
\node[check] (conflict) at (0,-3.9) {Xung đột?};
\node[check] (po) at (0,-5.2) {PO có D/D'?};
\node[check] (j) at (0,-6.5) {J rỗng?};
\node[box] (success) at (0,-7.8) {DETECTED\\trả lên mọi tầng};
% Cột phụ
\node[box] (abort) at (3.6,-1.3) {ABORTED\\trả lên mọi tầng};
\node[box] (back) at (3.6,-3.9) {Backtrack: khôi phục\\về lựa chọn trước};
\node[box] (cover) at (3.6,-6.5) {Chọn cover justify\\lưu trạng thái};
\node[check] (choices) at (7.4,-3.9) {Còn lựa chọn?};
\node[box] (retry) at (7.4,-2.3) {Thử cube tiếp theo\\kể cả PDCF còn lại};
\node[box] (fail) at (7.4,-5.5) {Hết ở mức gốc:\\UNTESTABLE};
\node[box] (drive) at (-3.6,-5.2) {Chọn PDC khả dụng\\lưu trạng thái};
% Luồng chính
\draw[->] (start)--(limit);
\draw[->] (limit)--node[lbl,right]{không}(imply);
\draw[->] (limit)--node[lbl,above]{có}(abort);
\draw[->] (imply)--(conflict);
\draw[->] (conflict)--node[lbl,right]{không}(po);
\draw[->] (conflict)--node[lbl,above]{có}(back);
\draw[->] (po)--node[lbl,right]{có}(j);
\draw[->] (po)--node[lbl,above]{không}(drive);
\draw[->] (j)--node[lbl,right]{có}(success);
\draw[->] (j)--node[lbl,above,pos=0.3]{không}(cover);
\draw[->] (back)--(choices);
\draw[->] (choices)--node[lbl,right]{có}(retry);
\draw[->] (choices)--node[lbl,right]{không}(fail);
% Vòng lặp quay lại kiểm tra giới hạn
\draw[->] (retry.north)--(7.4,0.75)--(-2.6,0.75)|-(limit.west);
\draw[->] (drive.north)|-(limit.west);
\draw[->] (cover.east)--(5.35,-6.5)--(5.35,-2.6)--(imply.east);
\node[lbl,font=\tiny,anchor=east] at (5.3,-5.3) {kiểm tra giới hạn};
% Thất bại nhánh khi không còn PDC
\draw[->,dashed] (drive.south)--(-3.6,-8.65)--(2.25,-8.65)--(2.25,-4.75)-|(back.south);
\node[font=\tiny,fill=white,inner sep=1pt] at (-0.9,-8.65) {Không có PDC khả dụng: thất bại nhánh};
\end{tikzpicture}
