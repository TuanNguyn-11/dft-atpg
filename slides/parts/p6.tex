\begin{frame}{P6 -- Kiểm chứng độc lập}
\small
\begin{itemize}
  \item Đọc \texttt{.bench} $\rightarrow$ fault list $\rightarrow$ PODEM (P5) $\rightarrow$
        mô phỏng lỗi độc lập $\rightarrow$ coverage, nén tập test.
  \item c17: 34 lỗi gốc (22 stem + 12 nhánh), gộp tương đương còn 22 đại diện.
  \item Mỗi pattern được mô phỏng lại ở mạch tốt và mạch lỗi; pattern có $X$ chỉ
        được xác nhận khi đã vét cạn mọi cách điền.
  \item Đầu vào sai (SA khác 0/1, nhánh không tồn tại, $k\le 0$) bị từ chối.
  \item Mạch tuần tự: $Q@0$ chưa biết; chỉ chuỗi bảo đảm với mọi trạng thái đầu
        mới tính là phát hiện.
\end{itemize}
\end{frame}
